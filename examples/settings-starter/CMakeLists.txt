cmake_minimum_required(VERSION 3.20)
project(MySettingsApp LANGUAGES CXX)

set(TESSERA_SOURCE_DIR "" CACHE PATH "Existing Tessera checkout")
set(TESSERA_IMPORT_MODE "add_subdirectory" CACHE STRING "add_subdirectory or FetchContent")
if(NOT EXISTS "${TESSERA_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set TESSERA_SOURCE_DIR to an existing Tessera checkout.")
endif()
# This example needs only the core and does not discover optional dependencies.
set(TESSERA_BUILD_FONTS OFF CACHE BOOL "" FORCE)
set(TESSERA_BUILD_VULKAN OFF CACHE BOOL "" FORCE)
if(TESSERA_IMPORT_MODE STREQUAL "add_subdirectory")
    add_subdirectory("${TESSERA_SOURCE_DIR}" tessera)
elseif(TESSERA_IMPORT_MODE STREQUAL "FetchContent")
    include(FetchContent)
    FetchContent_Declare(tessera SOURCE_DIR "${TESSERA_SOURCE_DIR}")
    FetchContent_MakeAvailable(tessera)
else()
    message(FATAL_ERROR "Use add_subdirectory or FetchContent for TESSERA_IMPORT_MODE.")
endif()

add_executable(settings_starter main.cpp)
target_link_libraries(settings_starter PRIVATE tessera::core)
if(WIN32)
    target_link_libraries(settings_starter PRIVATE user32 gdi32)
endif()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME settings_starter COMMAND settings_starter)
    if(WIN32)
        option(TESSERA_TEST_NATIVE_HOST "Run Settings native smoke (requires a desktop session)" OFF)
        if(TESSERA_TEST_NATIVE_HOST)
            add_test(NAME settings_starter_native COMMAND settings_starter --native-smoke)
            set_tests_properties(settings_starter_native PROPERTIES LABELS native TIMEOUT 60)
        endif()
    endif()
endif()
